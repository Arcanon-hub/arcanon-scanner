```rust
use std::collections::HashMap;
use std::path::{Path, PathBuf};

pub struct ServiceTarget {
    pub hostname: String,
    pub port: Option<u16>,
}

pub struct VariableStore {
    /// Highest priority: merged .env files (.env < .env.local < .env.development < .env.production)
    env_files: HashMap<String, String>,
    /// Middle priority: docker-compose environment entries (all services merged)
    compose_env: HashMap<String, String>,
    /// Lowest priority: k8s ConfigMap data entries (all ConfigMaps merged)
    k8s_env: HashMap<String, String>,
}

impl VariableStore {
    pub fn resolve(&self, key: &str) -> Option<&str> {
        self.env_files.get(key)
            .or_else(|| self.compose_env.get(key))
            .or_else(|| self.k8s_env.get(key))
            .map(|s| s.as_str())
    }

    pub fn resolve_to_target(&self, key: &str) -> Option<ServiceTarget> {
        let val = self.resolve(key)?;
        parse_url_to_service_target(val)
    }
}

pub fn build_variable_store(root: &Path, files: &[PathBuf]) -> VariableStore
```

.env file parsing logic (from RESEARCH.md Pattern 5):
```rust
fn parse_env_file(content: &str) -> HashMap<String, String> {
    let mut map = HashMap::new();
    for line in content.lines() {
        let line = line.trim();
        if line.is_empty() || line.starts_with('#') { continue; }
        let line = line.strip_prefix("export ").unwrap_or(line);
        if let Some((key, val)) = line.split_once('=') {
            let key = key.trim().to_string();
            let val = val.trim();
            let val = val.strip_prefix('"').and_then(|v| v.strip_suffix('"'))
                .or_else(|| val.strip_prefix('\'').and_then(|v| v.strip_suffix('\'')))
                .unwrap_or(val)
                .to_string();
            map.insert(key, val);
        }
    }
    map
}
```

.env merge priority function:
```rust
fn env_file_priority(path: &Path) -> u8 {
    match path.file_name().and_then(|n| n.to_str()).unwrap_or("") {
        ".env"             => 0,
        ".env.local"       => 1,
        ".env.development" => 2,
        ".env.production"  => 3,
        _                  => 4,  // other .env.* files get lowest priority
    }
}
```

Docker-compose parsing (from RESEARCH.md Pattern 6 — BOTH list and map forms required):
- List form: `environment: ["KEY=value", "KEY2=value2"]`
- Map form: `environment: { KEY: value, KEY2: value2 }`
- Use `serde_yaml_bw::Value` — NOT a typed struct (the compose schema has too much variation)

K8s ConfigMap parsing (from RESEARCH.md Pattern 7):
- Files can be multi-document (split on `\n---`)
- Deserialize each document into `K8sManifest { kind: Option<String>, data: Option<HashMap<String, String>> }`
- Only extract data where `kind == "ConfigMap"`

URL parsing for resolve_to_target:
- Use `url::Url::parse(val)` — add `url = "2"` to Cargo.toml
- Extract: `hostname = url.host_str().unwrap_or("").to_string()`
- Extract: `port = url.port()`
- If hostname is empty, return None (not a parseable URL)

Key pitfalls (from RESEARCH.md Pitfall 4, 5):
- Compose environment has two YAML forms — handle BOTH Value::Sequence and Value::Mapping
- .env merge order: .env → .env.local → .env.development → .env.production (LATER file wins for same key)
  Sort by priority number ASCENDING, then merge so each successive file overwrites earlier ones.
- Do NOT add `serde_yaml` — use `serde_yaml_bw` (already in Cargo.toml)

    .filter(|p| {
        p.file_name()
            .and_then(|n| n.to_str())
            .map_or(false, |n| n.starts_with(".env"))

// Sort by priority ascending (lower number = lower priority = loaded first = overwritten by later)
env_file_paths.sort_by_key(|p| env_file_priority(p));
let mut env_map = std::collections::HashMap::new();
for path in env_file_paths {
    if let Ok(content) = std::fs::read_to_string(path) {
        env_map.extend(parse_env_file(&content));

}
