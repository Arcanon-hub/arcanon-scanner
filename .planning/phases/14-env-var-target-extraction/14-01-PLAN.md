   ```rust
   /// Extract env var default value by scanning backward up to 20 lines.
   /// Returns "env:{VAR}" when no default is found, "" when VAR is not parseable.
   fn extract_env_default(lines: &[&str], line_idx: usize, language: &str) -> String {
       // Extract variable name from the matched line (line at line_idx)
       let matched_line = lines[line_idx];
       let var_name = match extract_first_string(matched_line) {
           Some(v) => v,
           None => return String::new(),
       };

       // Java @Value inline — no backward scan
       if language == "java" && matched_line.contains("@Value(") {
           // Pattern: @Value("${VAR:default}") — capture between ':' and '}'
           if let Some(dollar_pos) = matched_line.find("${") {
               let inner = &matched_line[dollar_pos + 2..];
               if let Some(colon_pos) = inner.find(':') {
                   if let Some(brace_pos) = inner[colon_pos + 1..].find('}') {
                       let default = &inner[colon_pos + 1..colon_pos + 1 + brace_pos];
                       let default = default.trim_matches('"').trim_matches('\'').trim();
                       if !default.is_empty() {
                           return default.to_string();
                       }
                   }
               }
           }
           return format!("env:{}", var_name);
       }

       // Tier-1 only languages: emit env hint without backward scan
       let tier1_only = matches!(language, "go" | "csharp");
       if tier1_only {
           return format!("env:{}", var_name);
       }

       // Also tier-1 only for Java System.getenv (not @Value)
       if language == "java" {
           return format!("env:{}", var_name);
       }

       // Backward scan: up to 20 lines before line_idx
       let scan_start = line_idx.saturating_sub(20);
       let window = &lines[scan_start..line_idx];

       for scan_line in window.iter().rev() {
           let trimmed = scan_line.trim();

           match language {
               "python" => {
                   // os.getenv("VAR", "default") or os.environ.get("VAR", "default")
                   if trimmed.contains(&format!("os.getenv(\"{}\"", var_name))
                       || trimmed.contains(&format!("os.getenv('{}',", var_name))
                       || trimmed.contains(&format!("os.environ.get(\"{}\"", var_name))
                       || trimmed.contains(&format!("os.environ.get('{}',", var_name))
                   {
                       // Extract second quoted string argument
                       if let Some(default) = extract_second_string_arg(trimmed) {
                           return default;
                       }
                   }
               }
               "typescript" | "javascript" => {
                   // process.env.VAR ?? "default" or process.env.VAR || "default"
                   if trimmed.contains(&format!("process.env.{}", var_name)) {
                       if let Some(default) = extract_after_nullish(trimmed) {
                           return default;
                       }
                   }
               }
               "rust" => {
                   // env::var("VAR").unwrap_or("default")
                   if trimmed.contains(&format!("env::var(\"{}\"", var_name))
                       || trimmed.contains(&format!("env::var('{}',", var_name))
                   {
                       if let Some(default) = extract_unwrap_or_arg(trimmed) {
                           return default;
                       }
                   }
               }
               "ruby" => {
                   // ENV.fetch("VAR", "default") or ENV["VAR"] || "default"
                   if trimmed.contains(&format!("ENV.fetch(\"{}\"", var_name))
                       || trimmed.contains(&format!("ENV.fetch('{}',", var_name))
                   {
                       if let Some(default) = extract_second_string_arg(trimmed) {
                           return default;
                       }
                   } else if trimmed.contains(&format!("ENV[\"{}\"]", var_name))
                       || trimmed.contains(&format!("ENV['{}']", var_name))
                   {
                       if let Some(default) = extract_after_or(trimmed) {
                           return default;
                       }
                   }
               }
               _ => {}
           }
       }

       // No default found — emit env hint
       format!("env:{}", var_name)
   }
   ```

   ```rust
   /// Extract first quoted string from a line (used to find variable/key names)
   fn extract_first_string(line: &str) -> Option<String> {
       let bytes = line.as_bytes();
       for i in 0..bytes.len() {
           let b = bytes[i];
           if b == b'"' || b == b'\'' {
               let quote = b as char;
               let rest = &line[i + 1..];
               if let Some(end) = rest.find(quote) {
                   return Some(rest[..end].to_string());
               }
           }
       }
       None
   }

   /// Extract second quoted string from a function call: fn("first", "second")
   fn extract_second_string_arg(line: &str) -> Option<String> {
       let bytes = line.as_bytes();
       let mut count = 0;
       let mut i = 0;
       while i < bytes.len() {
           let b = bytes[i];
           if b == b'"' || b == b'\'' {
               let quote = b as char;
               let rest = &line[i + 1..];
               if let Some(end) = rest.find(quote) {
                   count += 1;
                   if count == 2 {
                       return Some(rest[..end].to_string());
                   }
                   i += end + 2;
                   continue;
               }
           }
           i += 1;
       }
       None
   }

   /// Extract value after `??` or `||` operator, stripping quotes
   fn extract_after_nullish(line: &str) -> Option<String> {
       let pos = line.find("??").or_else(|| line.find("||"))?;
       let after = line[pos + 2..].trim();
       // Strip surrounding quotes
       for q in ['"', '\'', '`'] {
           if after.starts_with(q) {
               if let Some(end) = after[1..].find(q) {
                   return Some(after[1..1 + end].to_string());
               }
           }
       }
       None
   }

   /// Extract argument from `.unwrap_or("default")`
   fn extract_unwrap_or_arg(line: &str) -> Option<String> {
       let pos = line.find(".unwrap_or(")?;
       let after = &line[pos + 11..]; // len(".unwrap_or(") == 11
       let bytes = after.as_bytes();
       for (i, &b) in bytes.iter().enumerate() {
           if b == b'"' || b == b'\'' {
               let quote = b as char;
               let rest = &after[i + 1..];
               if let Some(end) = rest.find(quote) {
                   return Some(rest[..end].to_string());
               }
           }
       }
       None
   }

   /// Extract value after `||` operator (Ruby ENV["VAR"] || "default"), stripping quotes
   fn extract_after_or(line: &str) -> Option<String> {
       let pos = line.find("||")?;
       let after = line[pos + 2..].trim();
       for q in ['"', '\''] {
           if after.starts_with(q) {
               if let Some(end) = after[1..].find(q) {
                   return Some(after[1..1 + end].to_string());
               }
           }
       }
       None
   }
   ```
