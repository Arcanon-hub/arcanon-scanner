    resources expansion implementation:
    ```rust
    const RESOURCES_ROUTES: &[(&str, &str, &str)] = &[
        ("GET",    "/{name}",          "index"),
        ("POST",   "/{name}",          "create"),
        ("GET",    "/{name}/new",      "new"),
        ("GET",    "/{name}/:id",      "show"),
        ("GET",    "/{name}/:id/edit", "edit"),
        ("PUT",    "/{name}/:id",      "update"),
        ("DELETE", "/{name}/:id",      "destroy"),
    ];

    fn expand_resources(resource_name: &str) -> Vec<(String, String)> {
        let name = resource_name.trim_start_matches(':');
        RESOURCES_ROUTES.iter().map(|(method, path_tmpl, _)| {
            (method.to_string(), path_tmpl.replace("{name}", name))
        }).collect()
    }
    ```
