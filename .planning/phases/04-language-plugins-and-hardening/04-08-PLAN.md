    ```rust
    use std::path::PathBuf;
    use arcanon_scanner::core::scanner::{scan, ScanConfig};

    #[tokio::test]
    async fn test_polyglot_fixture_end_to_end() {
        let fixture_root = PathBuf::from(env!("CARGO_MANIFEST_DIR"))
            .join("tests/fixtures/polyglot");

        let config = ScanConfig {
            root: fixture_root,
            dry_run: true,
            hub_url: None,
            api_key: None,
            project_slug: Some("test".to_string()),
            output: None,
            plugins: None,
            excludes: vec![],
            repo_url: None,
            branch: None,
            commit_sha: None,
        };

        let payload = scan(config).await.expect("scan should succeed");

        // MONO-01/02: exactly 2 services detected (from 2 Dockerfiles)
        assert_eq!(payload.services.len(), 2,
            "Expected 2 services, got {}: {:?}",
            payload.services.len(),
            payload.services.iter().map(|s| &s.name).collect::<Vec<_>>());

        // Service names contain service-a and service-b
        let names: Vec<&str> = payload.services.iter().map(|s| s.name.as_str()).collect();
        assert!(names.iter().any(|n| n.contains("service-a")),
            "service-a not found in: {:?}", names);
        assert!(names.iter().any(|n| n.contains("service-b")),
            "service-b not found in: {:?}", names);

        // DETQ-05: NestJS two-phase endpoint
        let nestjs_endpoint = payload.endpoints.iter().find(|e|
            e.path == "/users/:id" && e.method == "GET"
        );
        assert!(nestjs_endpoint.is_some(),
            "NestJS GET /users/:id not found in: {:?}",
            payload.endpoints.iter().map(|e| (&e.method, &e.path)).collect::<Vec<_>>());

        // LPLU-02: FastAPI endpoint
        let fastapi_endpoint = payload.endpoints.iter().find(|e|
            e.path == "/items" && e.method == "GET"
        );
        assert!(fastapi_endpoint.is_some(),
            "FastAPI GET /items not found in endpoints");

        // Connections non-empty (httpx call in service-b detected)
        assert!(!payload.connections.is_empty(),
            "Expected at least one connection from httpx client call");
    }

    #[test]
    fn test_polyglot_fixture_files_exist() {
        // Quick sanity check that fixtures were created
        let root = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("tests/fixtures/polyglot");
        assert!(root.join("service-a/Dockerfile").exists());
        assert!(root.join("service-b/Dockerfile").exists());
        assert!(root.join("lib/shared.ts").exists());
        // lib/ must NOT have a Dockerfile (MONO-03 test requirement)
        assert!(!root.join("lib/Dockerfile").exists());
    }
    ```
