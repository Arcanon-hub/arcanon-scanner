```toml
# CLI

# Git context
gix = { version = "0.81", default-features = false, features = ["rev-parse-delegate", "interrupt"] }

# AST parsing — pin grammar crates to prevent ABI split (PITFALLS.md Pitfall 1)

# File walking
ignore = "0.4.25"
globset = "0.4.18"

# Serialization
serde = { version = "1.0.228", features = ["derive"] }
serde_json = "1.0.149"
toml = "1.1.2"
serde_yaml_bw = "2.5.4"  # replaces deprecated serde_yaml (PITFALLS.md Pitfall 2)

# HTTP upload (rustls only — no OpenSSL, musl-safe)
reqwest = { version = "0.13.2", default-features = false, features = ["rustls-tls", "json"] }

# Async runtime (reqwest only — NOT for plugins; plugins use rayon)
tokio = { version = "1.51", features = ["rt-multi-thread", "macros"] }

# CPU-bound parallelism for plugin execution
rayon = "1.11.0"

# Error handling
anyhow = "1.0.102"

# Logging
tracing = "0.1.44"
tracing-subscriber = { version = "0.3.23", features = ["env-filter"] }

[profile.release]
opt-level = "z"       # size-optimized (smaller than opt-level = 3)
lto = "fat"           # full LTO across all crates (most effective for binary size)
codegen-units = 1     # single codegen unit enables maximum LTO effectiveness
strip = "symbols"     # strip debug symbols explicitly (not strip = true shorthand)
```
