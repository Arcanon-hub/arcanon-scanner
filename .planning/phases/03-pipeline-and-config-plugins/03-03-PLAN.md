       use std::collections::HashMap;
       use crate::types::*;

       pub struct MergedResult {
           pub services: Vec<ServiceInfo>,
           pub endpoints: Vec<EndpointInfo>,
           pub connections: Vec<ConnectionInfo>,
           pub schemas: Vec<SchemaInfo>,
       }

       fn normalize_name(name: &str) -> String {
           name.to_lowercase()
               .replace(['_', ' '], "-")
       }

       fn service_priority(extraction_method: &str) -> u8 {
           match extraction_method {
               "compose" => 3,
               "package_json" => 2,
               "dockerfile" => 1,
               _ => 0,
           }
       }

       pub fn merge(results: Vec<ExtractionResult>) -> MergedResult {
           // Step 1: services
           let mut by_key: HashMap<String, ServiceInfo> = HashMap::new();
           for result in &results {
               for svc in &result.services {
                   let key = if svc.root_path.is_empty() {
                       normalize_name(&svc.name)
                   } else {
                       svc.root_path.clone()
                   };
                   by_key.entry(key)
                       .and_modify(|existing| {
                           if service_priority(&svc.extraction_method) > service_priority(&existing.extraction_method) {
                               existing.name = svc.name.clone();
                               existing.extraction_method = svc.extraction_method.clone();
                           }
                           if existing.language.is_empty() && !svc.language.is_empty() {
                               existing.language = svc.language.clone();
                           }
                           if existing.boundary_entry.is_none() {
                               existing.boundary_entry = svc.boundary_entry.clone();
                           }
                       })
                       .or_insert_with(|| svc.clone());
               }
           }
           let services: Vec<ServiceInfo> = by_key.into_values().collect();

           // Step 2: endpoints with spec-override
           let mut all_endpoints: Vec<EndpointInfo> = results.iter()
               .flat_map(|r| r.endpoints.iter().cloned())
               .collect();
           // Spec override: for same (service_name, method, path), spec wins
           // Strategy: partition into spec and non-spec; then keep non-spec only if no spec covers same key
           let (spec_eps, ast_eps): (Vec<_>, Vec<_>) = all_endpoints.drain(..)
               .partition(|ep| ep.extraction_method.starts_with("spec:"));
           let spec_keys: std::collections::HashSet<(String, String, String)> = spec_eps.iter()
               .map(|ep| (ep.service_name.clone(), ep.method.clone(), ep.path.clone()))
               .collect();
           let deduped_ast_eps: Vec<EndpointInfo> = ast_eps.into_iter()
               .filter(|ep| !spec_keys.contains(&(ep.service_name.clone(), ep.method.clone(), ep.path.clone())))
               .collect();
           let endpoints: Vec<EndpointInfo> = spec_eps.into_iter().chain(deduped_ast_eps).collect();

           // Step 3: connections (aggregate all)
           let connections: Vec<ConnectionInfo> = results.iter()
               .flat_map(|r| r.connections.iter().cloned())
               .collect();

           // Step 4: schemas with spec-override
           let all_schemas: Vec<SchemaInfo> = results.iter()
               .flat_map(|r| r.schemas.iter().cloned())
               .collect();
           let (spec_schemas, ast_schemas): (Vec<_>, Vec<_>) = all_schemas.into_iter()
               .partition(|s| s.extraction_method.starts_with("spec:"));
           let spec_schema_names: std::collections::HashSet<String> = spec_schemas.iter()
               .map(|s| s.name.clone())
               .collect();
           let deduped_ast_schemas: Vec<SchemaInfo> = ast_schemas.into_iter()
               .filter(|s| !spec_schema_names.contains(&s.name))
               .collect();
           let schemas: Vec<SchemaInfo> = spec_schemas.into_iter().chain(deduped_ast_schemas).collect();

           MergedResult { services, endpoints, connections, schemas }
       }
       ```

       use std::collections::HashMap;
       use crate::core::merger::MergedResult;

       pub fn normalize_path(path: &str) -> String {
           path.split('/')
               .map(|segment| {
                   if segment.starts_with(':') {
                       "{param}".to_string()
                   } else if segment.starts_with('{') && segment.ends_with('}') {
                       "{param}".to_string()
                   } else if segment == "*" {
                       "{*}".to_string()
                   } else {
                       segment.to_string()
                   }
               })
               .collect::<Vec<_>>()
               .join("/")
       }

       pub fn resolve(mut merged: MergedResult) -> MergedResult {
           // Build endpoint lookup: (METHOD_UPPER, normalized_path) → service_name
           let endpoint_lookup: HashMap<(String, String), String> = merged.endpoints
               .iter()
               .map(|ep| {
                   ((ep.method.to_uppercase(), normalize_path(&ep.path)), ep.service_name.clone())
               })
               .collect();

           // Update connections where we can match intra-repo
           for conn in &mut merged.connections {
               if let (Some(method), Some(path)) = (&conn.method, &conn.path) {
                   let key = (method.to_uppercase(), normalize_path(path));
                   if let Some(target_svc) = endpoint_lookup.get(&key) {
                       conn.target_name = target_svc.clone();
                   }
               }
           }

           merged
       }
       ```
