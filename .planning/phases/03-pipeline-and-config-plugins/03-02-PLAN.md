```rust
pub trait LanguagePlugin: Send + Sync {
    fn name(&self) -> &str;
    fn file_patterns(&self) -> &[&str];
    fn always_run(&self) -> bool { false }
    fn extract(&self, ctx: &ExtractionContext) -> ExtractionResult;
}

pub struct EndpointInfo {
    pub service_name: String,     // spec title or file stem
    pub method: String,           // HTTP method, "rpc", "query", "mutation", "subscription", "publish", "subscribe"
    pub path: String,             // /api/v1/users, ServiceName/MethodName, channel name
    pub handler: Option<String>,  // operation_id or rpc name
    pub kind: String,             // "rest", "grpc", "graphql", "asyncapi"
    pub confidence: Confidence,
    pub extraction_method: String, // "spec:openapi", "spec:proto", "spec:graphql", "spec:asyncapi"
}

pub struct SchemaInfo {
    pub name: String,
    pub role: String,             // "request", "response", "event", "type"
    pub file: Option<String>,     // source file relative path
    pub connection_ref: Option<String>,
    pub fields: Vec<FieldInfo>,
    pub confidence: Confidence,
    pub extraction_method: String,
}

pub struct FieldInfo {
    pub name: String,
    pub field_type: String,
    pub required: bool,
}
```
